public class MainActivity extends AppCompatActivity {
TextView name,mobile,email,address;
Button loadtData;
ProgressBar progressBar;
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);

        loadtData=findViewById(R.id.btn1);
        progressBar=findViewById(R.id.prog1);
        name=findViewById(R.id.txtload1);
        mobile=findViewById(R.id.txtload2);
        email=findViewById(R.id.txtload3);
        address=findViewById(R.id.txtload4);

        loadtData.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View view) {

                progressBar.setVisibility(View.VISIBLE);



                RequestQueue queue = Volley.newRequestQueue(MainActivity.this);
                String url = "http://192.168.0.109:8080/Apps/practice.json";

// Request a string response from the provided URL.
                StringRequest stringRequest = new StringRequest(Request.Method.GET, url,
                        new Response.Listener<String>() {
                            @Override
                            public void onResponse(String response) {
                                progressBar.setVisibility(View.GONE);

                                Log.d("serverRes",response);

                                try {
                                    JSONObject jsonObject=new JSONObject(response);
                                    String name1=jsonObject.getString("name");
                                    String mobile1=jsonObject.getString("mobile");
                                    String email1=jsonObject.getString("email");
                                    String address1=jsonObject.getString("address");

                                    name.setText(name1);
                                    mobile.setText(mobile1);
                                    email.setText(email1);
                                    address.setText(address1);

                                } catch (JSONException e) {
                                    throw new RuntimeException(e);
                                }

                            }
                        }, new Response.ErrorListener() {
                    @Override
                    public void onErrorResponse(VolleyError error) {
                        progressBar.setVisibility(View.GONE);
                        loadtData.setText("That didn't work!");
                    }
                });

// Add the request to the RequestQueue.
                queue.add(stringRequest);

            }
        });



    }
}
